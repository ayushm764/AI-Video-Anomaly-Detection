## Experiment 2: Lexical Analyzer (Token Separation)

```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>

const char *kw[] = {
    "int", "float", "double", "char",
    "if", "else", "for", "while", "return"
};

int isKW(char s[]) {
    int n = sizeof(kw) / sizeof(kw[0]);
    for (int i = 0; i < n; i++) {
        if (strcmp(s, kw[i]) == 0) return 1;
    }
    return 0;
}

int main() {
    char in[300], tok[50];
    int i = 0, j;
    printf("Enter a line of code:\n");
    fgets(in, sizeof(in), stdin);
    printf("\nToken Stream\n---------------------\n");
    while (in[i] != '\0') {
        if (isspace(in[i])) { i++; continue; }
        if (isalpha(in[i]) || in[i] == '_') {
            j = 0;
            while (isalnum(in[i]) || in[i] == '_') tok[j++] = in[i++];
            tok[j] = '\0';
            if (isKW(tok)) printf("Keyword -> %s\n", tok);
            else printf("Identifier -> %s\n", tok);
        }
        else if (isdigit(in[i])) {
            j = 0;
            while (isdigit(in[i])) tok[j++] = in[i++];
            tok[j] = '\0';
            printf("Number -> %s\n", tok);
        }
        else if (strchr("+-*/=%", in[i])) { printf("Operator -> %c\n", in[i]); i++; }
        else if (strchr(";,(){}", in[i])) { printf("Separator -> %c\n", in[i]); i++; }
        else { printf("Unknown -> %c\n", in[i]); i++; }
    }
    return 0;
}
```

**Sample Input:**
```
float avg_score = totalMarks + 15;
```

## Experiment 3: Symbol Table Construction

```cpp
#include <iostream>
#include <vector>
#include <iomanip>
using namespace std;

class Sym {
public:
    string id, dt, sc;
    int sz, ln;
    Sym(string i, string d, string s, int z, int l) {
        id = i; dt = d; sc = s; sz = z; ln = l;
    }
};

int main() {
    vector<Sym> tab;
    int opt;
    cout << "\n===== SYMBOL TABLE MANAGER =====\n";
    while (true) {
        cout << "\nSelect Operation";
        cout << "\n[1] Add New Symbol";
        cout << "\n[2] View Symbol Table";
        cout << "\n[0] Quit";
        cout << "\nEnter option: ";
        cin >> opt;
        if (opt == 1) {
            string id, dt, sc;
            int sz, ln;
            cout << "\nIdentifier Name : "; cin >> id;
            cout << "Data Type : "; cin >> dt;
            cout << "Scope Level : "; cin >> sc;
            cout << "Memory Size : "; cin >> sz;
            cout << "Declared at Line: "; cin >> ln;
            tab.push_back(Sym(id, dt, sc, sz, ln));
            cout << "\n>> Entry stored successfully.\n";
        }
        else if (opt == 2) {
            if (tab.empty()) { cout << "\n>> No symbols found.\n"; continue; }
            cout << "\n============================================================\n";
            cout << left << setw(5) << "#" << setw(15) << "Identifier"
                 << setw(12) << "Type" << setw(12) << "Scope"
                 << setw(8) << "Size" << setw(8) << "Line" << endl;
            cout << "============================================================\n";
            for (int i = 0; i < tab.size(); i++) {
                cout << left << setw(5) << i + 1 << setw(15) << tab[i].id
                     << setw(12) << tab[i].dt << setw(12) << tab[i].sc
                     << setw(8) << tab[i].sz << setw(8) << tab[i].ln << endl;
            }
            cout << "============================================================\n";
        }
        else if (opt == 0) { cout << "\nExiting Symbol Table Manager.\n"; break; }
        else { cout << "\nInvalid option. Try again.\n"; }
    }
    return 0;
}
```

```yacc
%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
int yylex();
void yyerror(const char *s);
struct node { char value[20]; struct node *left, *right; };
struct node* makeNode(char* val, struct node* l, struct node* r) {
    struct node* t = malloc(sizeof(struct node));
    strcpy(t->value, val); t->left = l; t->right = r; return t;
}
void printTree(struct node* root, int level) {
    if (root == NULL) return;
    for(int i=0;i<level;i++) printf("  ");
    printf("%s\n", root->value);
    printTree(root->left, level+1);
    printTree(root->right, level+1);
}
%}

%token <str> ID
%type <node> expr
%left '+' '-'
%left '*' '/'
%%
```

```yacc
#include <string.h>

%%
```

```yacc
    for(int i=0;i<firstCount;i++) if(first[i]==c) return;

    for(int i=0;i<followCount;i++) if(follow[i]==c) return;
    follow[followCount++]=c;

    if(visited[c]) return;
    visited[c]=1;
    if(!isupper(c)) { addFirst(c); return; }
    for(int i=0;i<6;i++) {
        if(production[i][0]==c) {
            char next=production[i][2];
            if(!isupper(next)) addFirst(next);
            else findFirst(next);
        }
    }
}
void findFollow(char c) {
    if(c=='E') addFollow('$');
    for(int i=0;i<6;i++) {
        for(int j=2;j<strlen(production[i]);j++) {
            if(production[i][j]==c) {
                if(production[i][j+1]!='\0') {
                    char next=production[i][j+1];
                    if(!isupper(next)) addFollow(next);
                    else {
                        firstCount=0; memset(visited,0,sizeof(visited));
                        findFirst(next);
                        for(int k=0;k<firstCount;k++) addFollow(first[k]);
                    }
                } else { if(c!=production[i][0]) findFollow(production[i][0]); }
            }
        }
    }
}

    int i, j;
    printf("\nFIRST SETS\n");
    for(i=0;i<3;i++) {
        firstCount=0; memset(visited,0,sizeof(visited));
```

## Experiment 9: LALR Parser

```c
#include <stdio.h>
#include <string.h>

char s[50], in[50];
int t=-1, i=0;

void p(char c) { s[++t]=c; }

void r() {
    int f=1;
    while(f) {
        f=0;
        if(s[t]=='b') { s[t]='A'; printf("Reduce A->b\n"); f=1; }
        else if(t>=1 && s[t]=='A' && s[t-1]=='a') {
            s[t-1]='A'; t--; printf("Reduce A->aA\n"); f=1;
        }
        else if(t>=1 && s[t]=='A' && s[t-1]=='A') {
            s[t-1]='S'; t--; printf("Reduce S->AA\n"); f=1;
        }
    }
}

int main() {
    printf("Enter input: ");
    scanf("%s", in);
    while(in[i]!='\0') {
        p(in[i]); printf("Shift %c\n", in[i]); i++; r();
    }
    r();
    if(t==0 && s[t]=='S') printf("\nAccepted\n");
    else printf("\nRejected\n");
    return 0;
}
```

## Experiment 11: Code Optimization

```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <ctype.h>

#define MAX 100

typedef struct {
    char left[10], right1[10], right2[10];
    char operator;
} Expression;

Expression expList[MAX];
char exprStore[MAX][30], varStore[MAX][10];
int expCount=0, storeCount=0;

int isNumeric(char *str) {
    for (int i=0; str[i]; i++) if(!isdigit(str[i])) return 0;
    return 1;
}
int evaluate(int a, int b, char op) {
    if(op=='+') return a+b; if(op=='-') return a-b;
    if(op=='*') return a*b; if(op=='/' && b!=0) return a/b;
    return 0;
}
int searchExpr(char *expr) {
    for(int i=0; i<storeCount; i++) if(strcmp(exprStore[i],expr)==0) return i;
    return -1;
}

int main() {
    int n;
    printf("Enter number of expressions: ");
    scanf("%d", &n);
    printf("Enter expressions (format: x = y op z):\n");
    for(int i=0; i<n; i++)
        scanf("%s = %s %c %s", expList[i].left, expList[i].right1,
              &expList[i].operator, expList[i].right2);

    printf("\nOptimized Code:\n");
    for(int i=0; i<n; i++) {
        char *l=expList[i].left, *r1=expList[i].right1, *r2=expList[i].right2;
        char op=expList[i].operator;

        if(strcmp(l,r1)==0 && strcmp(r2,"0")==0 && op=='+') continue;
        if(isNumeric(r1) && isNumeric(r2)) { printf("%s = %d\n",l,evaluate(atoi(r1),atoi(r2),op)); continue; }
        if(op=='+' && strcmp(r2,"0")==0) { printf("%s = %s\n",l,r1); continue; }
        if(op=='-' && strcmp(r2,"0")==0) { printf("%s = %s\n",l,r1); continue; }
        if(op=='*' && strcmp(r2,"1")==0) { printf("%s = %s\n",l,r1); continue; }
        if(op=='*' && strcmp(r2,"0")==0) { printf("%s = 0\n",l); continue; }
        if(op=='*' && strcmp(r2,"2")==0) { printf("%s = %s + %s\n",l,r1,r1); continue; }

        char buffer[30];
        sprintf(buffer,"%s %c %s",r1,op,r2);
        int pos=searchExpr(buffer);
        if(pos!=-1) printf("%s = %s\n",l,varStore[pos]);
        else {
            strcpy(exprStore[storeCount],buffer);
            strcpy(varStore[storeCount],l);
            storeCount++;
            printf("%s = %s\n",l,buffer);
        }
    }
    return 0;
}
```
